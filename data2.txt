1. WEB SERVERS & HOSTING PROJECTS
=======================================================================
# Python Local Web Server (Serves current folder on port 8000)
python -m http.server 8000
python3 -m SimpleHTTPServer 8080

# PHP Local Web Server (Serves current folder on port 8080)
php -S localhost:8080
php -S 0.0.0.0:8080 -t public/

# NodeJS Simple Web Server (requires 'npm install http-server')
npx http-server -p 8080
npx serve -s . -p 3000

# Expose Local Project to the Internet (Online Tunneling)
ngrok http 8080
ssh -R 80:localhost:8080 nokey@localhost.run
cloudflared tunnel --url http://localhost:8080

# Install and Run Nginx Web Server
pkg install nginx
nginx                  # Start server
nginx -s reload        # Reload config
nginx -s stop          # Stop server

5. CLONING & RUNNING GITHUB PROJECTS
=======================================================================
# General workflow for any open source project:
pkg install git
git clone https://github.com/username/project.git
cd project

# If it's a Python project:
python -m venv venv       # Create virtual environment
source venv/bin/activate  # Activate it
pip install -r requirements.txt  # Install dependencies
python main.py            # Run the project

# If it's a NodeJS project:
npm install               # Install dependencies
npm start                 # Run the project

# If it's a C/C++ project:
pkg install clang make
./configure
make
./executable_name

# Update an existing cloned project
git pull origin main

11. BACKGROUND EXECUTION (Keep projects running after closing Termux)
=======================================================================
# Method 1: NoHup (Standard Linux)
nohup python app.py > output.log 2>&1 &

# Method 2: Termux specific wake-lock (prevents Android from killing it)
termux-wake-lock
python app.py &
termux-wake-unlock

# Method 3: TMUX (Terminal Multiplexer)
pkg install tmux
tmux new -s myproject
# (Run your project inside)
# Press Ctrl+B, then D to detach
# To come back to it later:
tmux attach -t myproject
